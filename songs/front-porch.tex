% Example: a Nashville chart with guitar tab for the intro riff and solo.
% nnschart has no tab support (out of scope); the \tabstaff macro below is
% plain TikZ local to this song.
\documentclass{nnschart}
\songinfo{title={Front Porch}, artist={Example song}, key=G, tempo=96, time=4/4, feel={Train beat}}

% \tabstaff{bars}{slots per bar}{slot/string/fret, ...}
%   slot is 1-based across the whole staff; string 1 = high e, 6 = low E.
%   The staff wraps after every 2 bars; each line spans the column width.
\newlength\tabslot
\newcommand\tabstaff[3]{%
  \par
  \setlength\tabslot{\dimexpr\linewidth/(2*#2)\relax}%
  \pgfmathtruncatemacro\tablines{ceil(#1/2)}%
  \foreach \l in {1,...,\tablines}{%
    \pgfmathtruncatemacro\tabfirst{(\l-1)*2*#2}% slots before this line
    \pgfmathtruncatemacro\tablast{min(\tabfirst+2*#2, #1*#2)}%
    \pgfmathtruncatemacro\tabbars{(\tablast-\tabfirst)/#2}%
    \noindent
    \begin{tikzpicture}[x=\tabslot, y=-6.5pt,
                        fret/.style={fill=white, inner sep=0.8pt, font=\small\bfseries}]
      \useasboundingbox (0,0) rectangle (2*#2,7);
      \foreach \s in {1,...,6} \draw[black!55, line width=0.4pt] (0,\s) -- ++(\tabbars*#2,0);
      \node[overlay, font=\scriptsize\bfseries, align=center, inner sep=0pt]
        at ([xshift=-12pt]0,3.5) {T\\A\\B};
      \foreach \b in {0,...,\tabbars} \draw[line width=0.7pt] (\b*#2,1) -- (\b*#2,6);
      \foreach \t/\s/\f in {#3}{%
        \ifnum\t>\tabfirst\relax\ifnum\t>\tablast\relax\else
          \node[fret] at (\t-\tabfirst-0.5,\s) {\f};
        \fi\fi}
    \end{tikzpicture}\par\vspace{4pt}}}
